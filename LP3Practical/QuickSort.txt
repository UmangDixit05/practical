
import java.util.*;

public class QuickSort {

    // Deterministic Quick Sort
    static int partition(int arr[], int low, int high) {
        int pivot = arr[high];
        int i = low - 1;

        for (int j = low; j < high; j++) {
            if (arr[j] < pivot) {
                i++;
                int temp = arr[i];
                arr[i] = arr[j];
                arr[j] = temp;
            }
        }

        int temp = arr[i + 1];
        arr[i + 1] = arr[high];
        arr[high] = temp;

        return i + 1;
    }

    static void quickSort(int arr[], int low, int high) {
        if (low < high) {
            int p = partition(arr, low, high);
            quickSort(arr, low, p - 1);
            quickSort(arr, p + 1, high);
        }
    }

    // Randomized Quick Sort
    static int randomPartition(int arr[], int low, int high) {
        Random rand = new Random();
        int random = low + rand.nextInt(high - low + 1);

        int temp = arr[random];
        arr[random] = arr[high];
        arr[high] = temp;

        return partition(arr, low, high);
    }

    static void randomQuickSort(int arr[], int low, int high) {
        if (low < high) {
            int p = randomPartition(arr, low, high);
            randomQuickSort(arr, low, p - 1);
            randomQuickSort(arr, p + 1, high);
        }
    }

    public static void main(String[] args) {

        int arr1[] = {10, 7, 8, 9, 1, 5};
        int arr2[] = arr1.clone();

        quickSort(arr1, 0, arr1.length - 1);
        randomQuickSort(arr2, 0, arr2.length - 1);

        System.out.println("Deterministic Quick Sort:");
        System.out.println(Arrays.toString(arr1));

        System.out.println("Randomized Quick Sort:");
        System.out.println(Arrays.toString(arr2));
    }
}
